Skrip SQL Database Supabase
Salin dan jalankan skrip ini 1x di menu SQL Editor dashboard Supabase Anda.
📌 4 Langkah Cepat Menghubungkan Supabase:
  1. Buka dashboard supabase.com/dashboard lalu pilih project Anda.
  2. Di sidebar kiri, klik menu SQL Editor → klik New query.
  3. Klik tombol "Salin Seluruh Skrip SQL" di bawah, tempel (paste) ke SQL Editor, lalu klik tombol hijau Run.
  4. Buka menu Project Settings → API, salin Project URL dan anon public key, lalu tempel di halaman Pengaturan ExpediLog.
KODE SQL TABEL & KEBIJAKAN AKSES (RLS): 
-- ================================================================
-- SKRIP DATABASE EXPEDILOG TERPADU (SUPABASE POSTGRESQL)
-- Salin dan jalankan skrip ini 1x di menu SQL Editor dashboard Supabase Anda.
-- Mencakup: Resi Pengiriman, Catatan Tim, Portal Terpadu, & File Manager.
-- ================================================================

-- 1. TABEL RESI PENGIRIMAN
create table if not exists public.resi (
  id text primary key,
  nomor_resi text not null,
  nama_client text default '-',
  tanggal_kirim text default '',
  layanan text default 'Reguler',
  status text default 'Manifest',
  kurir text default '-',
  sla_rtn text default '-',
  sla_ev text default '-',
  drs_ke integer default 1,
  pengirim jsonb default '{}'::jsonb,
  penerima jsonb default '{}'::jsonb,
  barang jsonb default '{}'::jsonb,
  biaya numeric default 0,
  keterangan text default '',
  created_at bigint default (extract(epoch from now()) * 1000)::bigint,
  riwayat_status jsonb default '[]'::jsonb,
  updated_at timestamptz default now()
);

create index if not exists idx_resi_nomor on public.resi (nomor_resi);
create index if not exists idx_resi_status on public.resi (status);
create index if not exists idx_resi_kurir on public.resi (kurir);
create index if not exists idx_resi_client on public.resi (nama_client);
create index if not exists idx_resi_created on public.resi (created_at desc);

alter table public.resi enable row level security;
drop policy if exists "Akses Penuh ExpediLog Resi" on public.resi;
create policy "Akses Penuh ExpediLog Resi" on public.resi for all using (true) with check (true);

-- 2. TABEL CATATAN OPERASIONAL TIM (NOTEPAD TIM)
create table if not exists public.team_notes (
  id text primary key,
  title text not null default '',
  content text default '',
  author text default 'Tim Operasional',
  urgency text default 'Biasa',
  category text default 'Operasional',
  tags jsonb default '[]'::jsonb,
  created_at bigint default (extract(epoch from now()) * 1000)::bigint,
  updated_at bigint default (extract(epoch from now()) * 1000)::bigint
);

create index if not exists idx_notes_updated on public.team_notes (updated_at desc);
create index if not exists idx_notes_urgency on public.team_notes (urgency);

alter table public.team_notes enable row level security;
drop policy if exists "Akses Penuh ExpediLog Notes" on public.team_notes;
create policy "Akses Penuh ExpediLog Notes" on public.team_notes for all using (true) with check (true);

-- 3. TABEL PORTAL SISTEM TERPADU & TAUTAN OPERASIONAL
create table if not exists public.portal_systems (
  id text primary key,
  name text not null default '',
  url text not null default '',
  category text default 'Operasional',
  color text default 'emerald',
  description text default '',
  updated_at bigint default (extract(epoch from now()) * 1000)::bigint
);

create index if not exists idx_portal_cat on public.portal_systems (category);
create index if not exists idx_portal_updated on public.portal_systems (updated_at desc);

alter table public.portal_systems enable row level security;
drop policy if exists "Akses Penuh ExpediLog Portals" on public.portal_systems;
create policy "Akses Penuh ExpediLog Portals" on public.portal_systems for all using (true) with check (true);

-- 4. TABEL FILE MANAGER & CLOUD DRIVE
create table if not exists public.drive_files (
  id text primary key,
  name text not null default '',
  category text default 'document',
  ext text default '',
  size_bytes bigint default 0,
  uploaded_at bigint default (extract(epoch from now()) * 1000)::bigint,
  uploaded_by text default 'Pratama (Super Admin)',
  description text default '',
  data_url text default ''
);

create index if not exists idx_files_cat on public.drive_files (category);
create index if not exists idx_files_uploaded on public.drive_files (uploaded_at desc);

alter table public.drive_files enable row level security;
drop policy if exists "Akses Penuh ExpediLog Files" on public.drive_files;
create policy "Akses Penuh ExpediLog Files" on public.drive_files for all using (true) with check (true);

-- 5. AKTIFKAN REPLIKASI REALTIME WEBSOCKET (AMAN DIJALANKAN BERKALI-KALI)

  end;

Pengaturan WhatsApp Gateway

💡 Pengiriman Otomatis Senyap (Background): Dengan WhatsApp Gateway (seperti Fonnte / Wablas / Local REST), browser mengeksekusi pengiriman via REST API di latar belakang secara otomatis tanpa berpindah tab.
🏷️ Panduan Menghilangkan Watermark "| Sent via fonnte.com":
Teks watermark tersebut disisipkan langsung oleh server Fonnte khusus pada akun Free (Gratis).
• Opsi 1 (Fonnte Resmi): Upgrade akun Fonnte ke paket berbayar (mulai paket Regular). Watermark akan otomatis hilang 100%, atau bisa dikosongkan pada dashboard Fonnte (Menu Device > Edit > Kosongkan kolom Footer).
• Opsi 2 (100% Gratis Selamanya & Bebas Watermark): Pilih penyedia "Custom / Local REST API" (Gateway WhatsApp Web mandiri). Pesan dikirim langsung tanpa melalui server Fonnte sehingga 100% bebas watermark & gratis tanpa batas!
⚠️ Catatan Server Lokal: Opsi ini memerlukan program server gateway mandiri (seperti Node.js / Baileys) yang sedang aktif berjalan di port 3000 komputer Anda. Jika belum menjalankan server lokal di PC Anda, browser akan menolak koneksi (Failed to fetch). Untuk pengiriman langsung tanpa setup, silakan pilih Fonnte API.
